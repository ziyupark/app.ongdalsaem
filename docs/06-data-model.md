## 7. 목업 데이터 스키마 (권장)

```ts
type WeatherMood = 'clear'|'clear-night'|'cloudy'|'overcast'|'drizzle'|'rain'|'heavy-rain'|'shower'
  |'thunder'|'sleet'|'flurries'|'snow'|'heavy-snow'|'fog'|'dust'|'heat'|'cold'|'windy';
type Meal = 'breakfast'|'lunch'|'snack'|'dinner'|'late'|'dawn';

interface User { id: string; name: string; handle: string; city: string; bio: string;
  // name = DB profiles.display_name(1–30자, 중복 허용, null이면 1.3 미완료) · handle = profiles.username(a–z 0–9 . _ 3–20자, 유일)
  avatar: string; acornsReceived: number; dishCount: number; // avatar = profiles.avatar_url(Storage `avatars/<uid>/…`, 공개 읽기)
  handleAutoGenerated: boolean;   // 가입 때 자동 생성 후 아직 안 바꿨으면 true → 첫 멘션/공유 때 1회 제안 — DB profiles.handle_auto_generated
  handleChangedAt?: string }      // ISO, 30일 변경 제한 계산용 — DB profiles.handle_changed_at
// DB: supabase/migrations/20261106000000_profile_display_name_avatars.sql — profiles.display_name · handle_auto_generated · handle_changed_at · username 형식 검사 · Storage avatars 버킷
interface Dish { id: string; title: string; authorId: string; photos: string[]; story: string;
  weather: WeatherMood; tempC: number;              // 자동: 첫 사진 촬영 시각·위치의 날씨 (사용자 선택 없음)
  weatherSource: 'auto' | 'place-edited';           // 8.3d에서 장소를 바꿨으면 place-edited
  takenAt: string;                                  // 촬영 시각(EXIF, 없으면 공유 시각) — 날씨·Meal 기준
  moods: string[];                                 // 필수 1–3개(8.2 여러 개 선택): 기본 'cozy'|'calm'|'fresh'|'energized'|'comforting'|'nostalgic'|'celebrating'|'lazy'|'together'|'hearty' + 직접 쓴 무드(≤20자) — DB foods.mood_tags
  meals: Meal[];                                   // 필수 끼니 1개 이상(촬영 시각으로 1개 미리 선택) — DB foods.meal_type
  dietTags: ('vegan'|'seafood')[];                 // 8.2 Meal 줄의 Vegan · Seafood 칩(선택) — DB foods.meal_type에 함께 저장
  type: 'home'|'travel';                           // 8.2 Type. travel이면 ingredients·steps·kcalTotal 없음 — DB foods.dish_type
  // travel이면 설명 칸(story, ≤150자)이 "Where did you eat it?" 글이 된다. DB foods.place_name은 쓰지 않음(나중에 구조화된 장소용 예약, 항상 null)
  festive?: string;                                // 명절 태그(예: 'festive:halloween', 4.2 명절 헤더) — DB foods.character_tags에 함께 저장
  region: string; country: string; createdAt: string;
  placeNames?: Partial<Record<AppLanguage, { city: string; district?: string }>>; // 지역 이름을 9개 언어로(공유 · 장소 수정 때 저장) — 화면에는 보는 사람의 앱 언어로, 없으면 저장된 city · neighborhood. 표시 전용(필터 · 비교는 city · neighborhood) — DB foods.place_names(20261127000000)
  regionPrecision: 'neighborhood'|'city'|'country'; // 공유 시점의 설정 9.5 Shared precision 값(게시물마다 묻지 않음) — DB foods.region_precision
  acorns: number;          // 도토리(공개 반응) 수 — 피드 Actions · 사진 위 Acorn Tag · 프로필 통계. DB likes 테이블(이름은 그대로, 뜻 = 도토리)
  madeCount: number; madeByMe: boolean; // "I ate this" 기록 수 · 내가 기록했는지 — 7.1 하단 버튼. DB food_eaten_logs(후기 포함)
  commentCount: number; saved: boolean; acornGiven: boolean; // 내가 도토리를 줬는지(하트 없음)
  kcalTotal?: number;      // 작성자가 입력한 총 칼로리 추정치(선택), 표시는 "≈ 420 kcal"
  isPublic: boolean; allowComments: boolean;   // 8.2 Sharing 스위치 2개 — DB foods.visibility · foods.allow_comments(댓글 RLS에 반영)
  ingredients: { name: string; amount: string }[]; steps: { text: string; timerSec?: number }[];
  tags: string[] }
interface Place { id: string; name: string; dish: string; priceKRW: number; distanceM: number;
  walkMin: number; openUntil: string; acorns: number; note: string; address: string;
  addressKo: string; lat: number; lng: number; phone?: string }
interface Forecast { city: string; mood: WeatherMood; tempC: number; hi: number; lo: number;
  condition: string; until?: string; hourly: { meal: Meal; tempC: number }[] }
interface Timer { id: string; label: string; totalSec: number; remainingSec: number;
  running: boolean; dishId?: string; step?: number }
interface Settings { appearance: 'system'|'light'|'dark'; tempUnit: 'C'|'F';   // 9.1 선택 시트
  language: string; useDeviceLanguage: boolean;    // 9.6 — BCP 47('en-US','ko' …). useDeviceLanguage=true면 기기 언어 — DB profiles.language · use_device_language
  defaultPublic: boolean; defaultAllowComments: boolean; // 9.5 Who sees your new dishes → 8.2 Sharing 기본값 — DB profiles.default_visibility · default_allow_comments
  useLocation: boolean; regionPrecision: 'neighborhood'|'city'|'country'; fallbackCity?: string; // 9.5 · 9.7(위치 꺼짐일 때만) — DB profiles.location_enabled · shared_precision
  notify: { acorns: boolean; comments: boolean; followers: boolean;
    dailyPick: boolean; dailyPickTime: string; timer: boolean };
  /* marketing: v1 미사용(광고 없음) — DB 컬럼만 있음 */ }
// 측정 단위·식단 선호 설정은 두지 않는다.
// DB: supabase/migrations/20261030000000_dish_type_privacy_language.sql — foods(dish_type · place_name(예약) · allow_comments · region_precision) · profiles(언어 · 공개 기본값 · 메시지 · 위치)
// Today's pick · 끼니 슬롯은 서버 API 없이 앱이 공개 게시물 풀에서 고른다(src/services/recommend.ts, 규칙 = 05-behavior §6.1 · §6.2). Mood Prompt(기분 입력)는 없다.
// 취향: profiles.favorite_tags(text[]) — 1.3b Taste에서 저장, Today's pick 취향 보너스에 사용
// 매일 추천 알림 시간: 기기 prefs.pickTime("HH:MM", 기본 07:30, 1.5에서 정함) — 푸시 토큰과 함께 서버에 저장(20261108000000_daily_pick_push.sql)
interface Notification { id: string; type: 'acorn'|'comment'|'follow'|'daily-pick'|'timer'|'moderation';
  actorIds: string[];            // 묶인 사람들(도토리 "A and N others")
  dishId?: string; commentId?: string; threadId?: string; timerId?: string;
  text: string; createdAt: string; readAt?: string }   // 30일 보관
// ── 메시지(DM) — 4.11 · 4.5 Message에서 씀 ──
interface Message { id: string; threadId: string; senderId: string; text: string; createdAt: string; seenAt?: string }
interface Thread { id: string; memberIds: [string, string]; requestFor?: string; lastMessageAt: string } // requestFor = 아직 수락(답장) 안 한 쪽
// DB: supabase/migrations/20261029000000_follows_and_messages.sql — follows · message_threads · messages (+ start_thread(other) RPC, 차단 시 팔로우·대화 차단)
// ── 컬렉션 — 6.2 Saved에서 씀(만들기 · 담기 · 길게 눌러 삭제) ──
interface Collection { id: string; ownerId: string; name: string; dishIds: string[]; createdAt: string; updatedAt: string } // 7.6 · 6.2, "All saved"는 saves 전체(가상)
// DB: supabase/migrations/20261029010000_save_collections.sql — collections(이름 1–30자, 주인별 중복 금지) · collection_items(collection × save, 저장 해제 시 함께 삭제), 본인만 읽기/쓰기. 날씨 그룹은 저장하지 않고 foods 날씨로 계산
interface Report { id: string; reporterId: string; foodId?: string; commentId?: string; reportedUserId?: string;
  reason: 'sexual'|'child_safety'|'harassment'|'hate_violence'|'copyright'|'spam'|'unsafe_recipe'|'off_topic'|'other'; // 7.3 순서
  note?: string;                                   // ≤200자
  status: 'open'|'actioned'|'dismissed'; createdAt: string; resolvedAt?: string }
interface ModerationAction { id: string; foodId?: string; commentId?: string; targetUserId: string;
  action: 'keep'|'hide'|'remove'|'warn'|'suspend'|'ban'|'restore'; reason: Report['reason']; note?: string;
  suspendedUntil?: string; notifiedAt?: string; appealStatus: 'none'|'requested'|'upheld'|'reversed'; createdAt: string }
// Dish에 moderationState: 'visible'|'hidden_pending'|'removed' (DB foods.moderation_state — 작성자 본인에게만 보이는 상태가 있음)
// DB: supabase/migrations/20261030010000_moderation.sql — reports 사유·메모·상태, 신고 즉시 신고자 숨김(hidden_foods), 3명↑ 또는 sexual/child_safety → 자동 임시 숨김, moderation_actions(작성자 읽기 + Appeal 요청), moderation_queue(운영자 전용, 24h 기한)
interface Table { id: string; slug: string; name: string;     // '#SoloMeals'
  description?: string; rules?: string; cover?: string; createdBy?: string; // 없으면 ongdalsaem(운영자)
  memberCount: number; postsToday: number; joined: boolean }    // 4.7 · 4.8 — DB topic_tables · table_members · table_posts(멤버가 자기 요리만)
interface FeedItem { dishId: string; source: 'person'|'table'; tableId?: string } // 4.4 Following 피드 — filter: 'all'|'people'|'tables'(기기에 마지막 선택 저장). source='table'이면 "from #테이블" 줄
interface Question { id: string; authorId: string; body: string;  // 5–200자
  category: 'where_to_eat'|'recipe_help'|'ingredients'|'market_deals'|'other';
  photo?: string; neighborhood?: string; city?: string; country: string; tableId?: string; // 테이블 안 질문이면 tableId
  regionPrecision: 'neighborhood'|'city'|'country'; // 질문한 사람의 9.5 Shared precision — 그보다 자세한 장소 이름은 저장하지 않음(neighborhood/city가 null)
  bestAnswerId?: string; answerCount: number; meTooCount: number; meToo: boolean;
  expiresAt: string; createdAt: string }                        // 7일 — 3.1 · 3.7 · 3.8
interface Answer { id: string; questionId: string; authorId: string; body: string; // ≤500자
  helpfulCount: number; helpful: boolean; createdAt: string }
interface Challenge { id: string; slug: string; title: string; description?: string; rule?: string;
  tag: string;                                     // 'MidnightRamyeon' → 게시물 태그 #MidnightRamyeon
  country?: string; cover?: string; startsAt: string; endsAt: string;
  participantCount: number; entryCount: number;
  phase: 'upcoming'|'running'|'closing_soon'|'ended'; // closing_soon = 72시간 미만
  joined: boolean; myEntryIds: string[] }          // 4.1 카드 · 4.10 · 4.10b
interface ChallengeEntry { challengeId: string; dishId: string; userId: string; createdAt: string } // 기간 안 · 참여자 · 본인 공개 게시물
// DB: supabase/migrations/20261030020000_tables_questions_challenges.sql — topic_tables(기본 6개) · table_members · table_posts · questions · answers · question_me_too · answer_helpful · challenges · challenge_participants · challenge_entries · view challenge_cards(phase · joined), 신고 대상에 question_id · answer_id 추가. 수는 트리거가 관리
// DB: supabase/migrations/20261105000000_question_area_privacy.sql — questions.region_precision, question_areas(약 5 km geohash, 클라이언트 읽기 불가), ask_question()(질문 작성은 이것만) · nearby_questions(geohash)(약 20 km 범위, 차단 · 만료 제외)
// Eat together는 2단계 — 데이터 모델 없음
interface SupportTicket { id: string; userId: string; topic: 'bug'|'account'|'report_content'|'other';
  message: string; screenshot?: string; replyEmail?: string;   // 5–1000자, 답장 = 로그인 이메일
  appVersion: string; os: string; language: string;            // 자동 첨부
  status: 'open'|'answered'|'closed'; createdAt: string; closedAt?: string } // 9.8a — 처리 완료 후 3년 보관
// DB: supabase/migrations/20261105010000_support_and_marketing.sql — support_tickets(본인만 insert · select) · (v1 미사용) profiles.marketing_ads · marketing_night_ads · marketing_consented_at(야간은 광고 동의가 있어야 켤 수 있음) · marketing_consent_log(바꿀 때마다 자동 기록, 9.9e "You agreed on …"과 2년 재확인용)
// DB: supabase/migrations/20261111000000_nearby_questions_wide.sql — nearby_questions_wide(geohash)(한 단계 넓은 3자리 칸, 규칙은 nearby_questions와 같음) — Our area 질문 채우기
// Our area "Popular today"는 테이블 없이 foods(오늘 0시 이후 · geohash6 · neighborhood · city)를 앱에서 이름별로 묶어 계산
// 냉장고 메모(6.3 Fridge)는 기기에만 저장(AsyncStorage, { name, zone: 'fridge'|'freezer', addedAt }) — DB 없음. "What you can cook now"는 foods.ingredients가 있는 최신 공개 게시물 200개와 비교
// 식단 계획(6.3 이번 주 한 줄 · 7.1 Add to meal plan)도 기기에만 저장(AsyncStorage `ongdalsaem.mealPlan`) — DB 없음, 동기화 안 함. 30일 지난 계획은 불러올 때 지움
interface PlanItem { id: string; date: string /* 로컬 YYYY-MM-DD */; meal: 'lunch'|'dinner'; foodId?: string; name: string; photo?: string } // 하루 끼니당 1개(같은 끼니는 교체)
// 주방 타이머(5.x)도 기기에만 저장
interface Follow { followerId: string; followeeId: string; createdAt: string }
interface Country { code: string; name: string; city: string; localTime: string; mood: WeatherMood;
  tempC: number; dishCount7d: number }
```
